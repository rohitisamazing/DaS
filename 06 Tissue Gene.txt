import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import confusion_matrix
import numpy as np

# --------------------------------------------------------------
# ✅ Load dataset
# --------------------------------------------------------------
df = pd.read_csv("tissue_gene_expression.csv")

# features = all gene columns
X = df.drop("tissue", axis=1)

# actual classes
y_true = df["tissue"]

# --------------------------------------------------------------
# ✅ Standardize data (important for k-means)
# --------------------------------------------------------------
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# --------------------------------------------------------------
# ✅ K-Means with K = 7
# --------------------------------------------------------------
k = 7
kmeans = KMeans(n_clusters=k, n_init=10, random_state=42)
clusters = kmeans.fit_predict(X_scaled)

# --------------------------------------------------------------
# ✅ Comparison table (cluster vs tissue type)
# --------------------------------------------------------------
comparison = pd.crosstab(clusters, y_true)
print("\nCluster vs Actual Tissue Table:\n")
print(comparison)

# --------------------------------------------------------------
# ✅ Run algorithm multiple times
# --------------------------------------------------------------
print("\nRunning KMeans multiple times to observe variation:\n")
for i in range(5):
    km = KMeans(n_clusters=7, n_init=10)
    labels = km.fit_predict(X_scaled)
    print(f"Run {i+1}: Cluster Distribution = {np.bincount(labels)}")
